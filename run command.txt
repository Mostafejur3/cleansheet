Windows PowerShell / CMD:
  py -m venv .venv
  .venv\Scripts\activate
  python -m pip install --upgrade pip
  python -m pip install -r requirements.txt
  python -m streamlit run app.py

macOS / Linux:
  python3 -m venv .venv
  source .venv/bin/activate
  python -m pip install --upgrade pip
  python -m pip install -r requirements.txt
  python -m streamlit run app.py

Run tests:
  python -m pytest -q
